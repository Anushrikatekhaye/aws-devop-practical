name: Terraform CI/CD

on:
  pull_request:
    paths: ["terraform/**", "app/**", ".github/workflows/terraform.yml"]
  push:
    branches: [main]
    paths: ["terraform/**", "app/**", ".github/workflows/terraform.yml"]
  workflow_dispatch:
    inputs:
      destroy:
        description: "Run terraform destroy instead of apply"
        type: boolean
        default: false

permissions:
  id-token: write # OIDC to AWS (no long-lived keys)
  contents: read
  pull-requests: write

env:
  TF_IN_AUTOMATION: "true"
  AWS_REGION: ${{ vars.AWS_REGION || 'ap-south-1' }}
  TF_VAR_domain_name: ${{ vars.DOMAIN_NAME || '' }}
  TF_VAR_route53_zone_id: ${{ vars.ROUTE53_ZONE_ID || '' }}

defaults:
  run:
    working-directory: terraform

jobs:
  validate:
    name: Validate
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: hashicorp/setup-terraform@v3
        with:
          terraform_version: 1.9.5

      - name: Format check
        run: terraform fmt -check -recursive

      - name: Init (no backend)
        run: terraform init -backend=false

      - name: Validate
        run: terraform validate

      - name: Lint Python
        working-directory: app
        run: |
          pip install flake8
          flake8 app.py --max-line-length 100

  plan:
    name: Plan
    needs: validate
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: hashicorp/setup-terraform@v3
        with:
          terraform_version: 1.9.5

      - uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: ${{ secrets.AWS_ROLE_ARN }}
          aws-region: ${{ env.AWS_REGION }}

      - name: Init
        run: |
          terraform init \
            -backend-config="bucket=${{ secrets.TF_STATE_BUCKET }}" \
            -backend-config="key=devops-practical/${{ vars.ENVIRONMENT || 'dev' }}/terraform.tfstate" \
            -backend-config="region=${{ env.AWS_REGION }}" \
            -backend-config="dynamodb_table=${{ secrets.TF_LOCK_TABLE }}" \
            -backend-config="encrypt=true"

      - name: Plan
        run: terraform plan -input=false -out=tfplan

      - name: Upload plan
        uses: actions/upload-artifact@v4
        with:
          name: tfplan
          path: terraform/tfplan
          retention-days: 3

  apply:
    name: Apply
    needs: plan
    if: github.ref == 'refs/heads/main' && github.event_name != 'pull_request' && !inputs.destroy
    runs-on: ubuntu-latest
    environment: production # configure required reviewers for a manual approval gate
    steps:
      - uses: actions/checkout@v4

      - uses: hashicorp/setup-terraform@v3
        with:
          terraform_version: 1.9.5

      - uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: ${{ secrets.AWS_ROLE_ARN }}
          aws-region: ${{ env.AWS_REGION }}

      - uses: actions/download-artifact@v4
        with:
          name: tfplan
          path: terraform

      - name: Init
        run: |
          terraform init \
            -backend-config="bucket=${{ secrets.TF_STATE_BUCKET }}" \
            -backend-config="key=devops-practical/${{ vars.ENVIRONMENT || 'dev' }}/terraform.tfstate" \
            -backend-config="region=${{ env.AWS_REGION }}" \
            -backend-config="dynamodb_table=${{ secrets.TF_LOCK_TABLE }}" \
            -backend-config="encrypt=true"

      - name: Apply
        run: terraform apply -input=false tfplan

      - name: Capture outputs
        id: out
        run: echo "url=$(terraform output -raw application_url)" >> "$GITHUB_OUTPUT"

      - name: Health check
        run: |
          echo "Checking ${{ steps.out.outputs.url }}/health"
          for i in $(seq 1 30); do
            code=$(curl -sk -o /dev/null -w '%{http_code}' "${{ steps.out.outputs.url }}/health" || true)
            echo "attempt $i -> HTTP $code"
            [ "$code" = "200" ] && exit 0
            sleep 20
          done
          echo "Health check failed" && exit 1

      - name: DB connectivity check
        run: |
          curl -skf "${{ steps.out.outputs.url }}/db" | tee /dev/stderr | grep -q '"database":"connected"'

  destroy:
    name: Destroy
    if: inputs.destroy
    runs-on: ubuntu-latest
    environment: production
    steps:
      - uses: actions/checkout@v4

      - uses: hashicorp/setup-terraform@v3
        with:
          terraform_version: 1.9.5

      - uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: ${{ secrets.AWS_ROLE_ARN }}
          aws-region: ${{ env.AWS_REGION }}

      - name: Init
        run: |
          terraform init \
            -backend-config="bucket=${{ secrets.TF_STATE_BUCKET }}" \
            -backend-config="key=devops-practical/${{ vars.ENVIRONMENT || 'dev' }}/terraform.tfstate" \
            -backend-config="region=${{ env.AWS_REGION }}" \
            -backend-config="dynamodb_table=${{ secrets.TF_LOCK_TABLE }}" \
            -backend-config="encrypt=true"

      - name: Destroy
        run: terraform destroy -input=false -auto-approve
